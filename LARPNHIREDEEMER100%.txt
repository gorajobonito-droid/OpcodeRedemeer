local function trim(s) return (tostring(s or ""):gsub("^%s+",""):gsub("%s+$","")) end
local function basicClean(s)
    s = tostring(s or ""):lower():gsub("[%?%!%.%,%;%:\"'`]"," "):gsub("%s+"," ")
    return trim(s)
end
local function upperClean(s)
    return (tostring(s or ""):upper():gsub("%s+",""):gsub("[%?%.%,!\"'`\n\r]",""))
end
local function ordinal(n)
    return ({[1]="1st",[2]="2nd",[3]="3rd"})[n] or (tostring(n).."th")
end
local function applyAliases(s)
    local out = {}
    for word in s:gmatch("%S+") do
        local sw = WORD_ALIAS[word]
        if sw ~= nil then if sw ~= "" then out[#out+1]=sw end
        else out[#out+1]=word end
    end
    return table.concat(out," ")
end
local function stripFillers(s)
    local out = " "..s.." "
    for _,f in ipairs(FILLERS) do
        out = out:gsub("%s"..f:gsub("%%","%%%%").."%s"," ")
    end
    out = out:gsub("%s+my%s+"," "):gsub("^%s*my%s+"," ")
    out = out:gsub("%s+sammys?%s+"," "):gsub("^%s*sammys?%s+"," ")
    out = out:gsub("%s+the%s+"," "):gsub("%s+of%s+"," "):gsub("%s+for%s+"," ")
    out = out:gsub("%s+a%s+"," "):gsub("%s+an%s+"," "):gsub("%s+in%s+"," ")
    out = out:gsub("%s+i%s+"," "):gsub("%s+is%s+"," "):gsub("%s+was%s+"," ")
    return trim((out:gsub("%s+"," ")))
end
local function toDigits(s)
    local conv = s
    for word,n in pairs(NUMWORDS) do
        conv = conv:gsub("%f[%a]"..word.."%f[%A]", ordinal(n))
    end
    for word,n in pairs(CARDINALS) do
        conv = conv:gsub("%f[%a]"..word.."%f[%A]", tostring(n))
    end
    return conv
end
local function multiplierOf(text)
    local l = text:lower()
    if l:find("twice") then return 2 end
    if l:find("thrice") then return 3 end
    local n = tonumber(l:match("(%d+)%s*times"))
    if n and n>=2 and n<=10 then return n end
    for word,value in pairs(CARDINALS) do
        if value>=2 and value<=10 and l:find("%f[%a]"..word.."%s+times") then return value end
    end
    return nil
end
local function stripMultiplier(text)
    local s = text
    s = s:gsub("%s*%d+%s*times%s*"," ")
    s = s:gsub("%s*twice%s*"," ")
    s = s:gsub("%s*thrice%s*"," ")
    for word,value in pairs(CARDINALS) do
        if value>=2 and value<=10 then s = s:gsub("%s*"..word.."%s+times%s*"," ") end
    end
    return trim(s) ~= "" and s or text
end

local PROTECTED = {["black and white"]="\1bw\1"}
local UNPROTECT = {["\1bw\1"]="black and white"}
local function splitQuestions(text)
    local s = " "..tostring(text or ""):lower().." "
    for phrase,tok in pairs(PROTECTED) do s = s:gsub(phrase,tok) end
    s = s:gsub("%s*[%?%;%,]%s*"," and ")
    s = s:gsub("%s*&%s*"," and "):gsub("%s*%+%s*"," and ")
    s = " "..basicClean(s).." "
    s = s:gsub("%s+add%s+"," and "):gsub("%s+plus%s+"," and "):gsub("%s+then%s+"," and ")
    local parts = {}
    for part in (s.." and "):gmatch("(.-)%s+and%s+") do
        part = trim(part)
        if part ~= "" then
            for tok,ph in pairs(UNPROTECT) do part = part:gsub(tok,ph) end
            parts[#parts+1]=part
        end
    end
    if #parts==0 then parts[1]=trim(basicClean(text)) end
    return parts
end

local function findRarestExist()
    local minVal, names = math.huge, {}
    for key, val in pairs(EXIST_DB) do
        local num = tonumber(val:gsub("[^%d]",""))
        if num and num>0 and num<minVal then minVal=num; names={key}
        elseif num and num>0 and num==minVal then names[#names+1]=key end
    end
    if #names==0 then return nil end
    local out = {}
    for _,n in ipairs(names) do out[#out+1]=n:upper() end
    return table.concat(out,"AND")
end

local function dynamicAnswer(text)

        return MONTHS_UP[now.month]..tostring(now.day)..tostring(now.year)
    end
    if q=="today day of week" or q=="what day is it" or q=="current day" then
        return DAYS_UP[now.wday]
    end
    if q=="today day of month" or q=="day of month" or q=="today date of the month"
        or q=="date of the month" then
        return tostring(now.day)
    end
    if q=="current month" or q=="this month" then return MONTHS_UP[now.month] end
    if q=="current year" or q=="this year" then return tostring(now.year) end
    return nil
end

local function localAnswer(text)
    if not text or text == "" then return nil end
    local dyn = dynamicAnswer(text); if dyn then return dyn end
    local mp = tryMultiplierPreview(text); if mp then return mp end
    local ex = tryExistLookup(text); if ex then return ex end
    local raw = basicClean(text)
    local aliased = applyAliases(raw)
    local core = stripFillers(aliased)
    local candidates = {raw, aliased, core}
    local base = #candidates
    for i=1,base do
        local conv = toDigits(candidates[i])
        if conv ~= candidates[i] then candidates[#candidates+1]=conv end
    end
    for _,k in ipairs(candidates) do
        k = trim(k)
        if k~="" and SAB_DB[k] then return SAB_DB[k] end
    end
    return nil
end

local function simpleLookup(text)
    if not text or text == "" then return nil end
    local raw = basicClean(text)
    local aliased = applyAliases(raw)
    local core = stripFillers(aliased)
    local candidates = {raw, aliased, core}
    local base = #candidates
    for i=1,base do
        local conv = toDigits(candidates[i])
        if conv ~= candidates[i] then candidates[#candidates+1]=conv end
    end
    for _,k in ipairs(candidates) do
        k = trim(k)
        if k~="" and SAB_DB[k] then return SAB_DB[k] end
    end
    return nil
end

local function trySubSplit(part)
    if not part or part == "" then return nil end
    local tokens = {}
    for w in part:gmatch("%S+") do tokens[#tokens+1] = w end
    local n = #tokens
    if n < 2 then return nil end
    for i = 1, n - 1 do
        local a = table.concat(tokens, " ", 1, i)
        local b = table.concat(tokens, " ", i + 1, n)
        if localAnswer(a) and localAnswer(b) then return {a, b} end
    end
    return nil
end

local function fuzzyResolvePhrase(part)
    if not part or part == "" then return nil end
    local baseClean = basicClean(part)
    if baseClean == "" then return nil end
    local tokens = {}
    for w in baseClean:gmatch("%S+") do tokens[#tokens+1] = w end
    local n = #tokens
    if n == 0 or n > 80 then return nil end
    local answers = {}
    local matchedTokens = 0
    local i = 1
    while i <= n do
        local matched = false
        local maxLen = math.min(n - i + 1, 8)
        for len = maxLen, 1, -1 do
            local candidate = table.concat(tokens, " ", i, i + len - 1)
            local ans = simpleLookup(candidate)
            if not ans then ans = tryMultiplierPreview(candidate) end
            if not ans then ans = dynamicAnswer(candidate) end
            if ans and ans ~= "" then
                local up = upperClean(ans)
                if up ~= "" then
                    answers[#answers+1] = up
                    matchedTokens = matchedTokens + len
                    i = i + len
                    matched = true
                    break
                end
            end
        end
        if not matched then i = i + 1 end
    end
    if #answers == 0 then return nil end
    return answers, matchedTokens, n
end

local function resolvePart(part, allowMul)
    if not part or part=="" then return nil end
    local times = allowMul and multiplierOf(part) or nil
    local q = allowMul and stripMultiplier(part) or part
    q = trim(q)
    if q=="" then return nil end
    local onlyNum = q:match("^%d+$")
    if onlyNum then return times and string.rep(onlyNum,times) or onlyNum end
    local ans = localAnswer(q)
    if not ans then return nil end
    local clean = upperClean(ans)
    if clean=="" then return nil end
    return times and string.rep(clean,times) or clean
end

local function solveRiddle(text)
    if not text or text=="" then return nil,nil,"empty" end
    local parts = splitQuestions(text)
    local answers, breakdown = {}, {}
    local missing = 0
    for _, part in ipairs(parts) do
        local ok, result = pcall(resolvePart, part, cfg.multiplier)
        if ok and result and result~="" then
            answers[#answers+1]=result
            breakdown[#breakdown+1]=part:sub(1,30).." = "..result
        else
            local fz, matched, total
            pcall(function() fz, matched, total = fuzzyResolvePhrase(part) end)
            if fz and #fz > 0 then
                for _, a in ipairs(fz) do answers[#answers+1] = a end
                if matched and total and matched < total then
                    breakdown[#breakdown+1] = "fuzzy("..matched.."/"..total.."): "
                        ..table.concat(fz,"+")
                else
                    breakdown[#breakdown+1] = "fuzzy: "..table.concat(fz,"+")
                end
            else
                local subs = trySubSplit(part)
                if subs then
                    local okAll = true
                    local tmp = {}
                    for _, sub in ipairs(subs) do
                        local sr = resolvePart(sub, cfg.multiplier)
                        if not sr or sr == "" then okAll=false; break end
                        tmp[#tmp+1] = {q=sub, a=sr}
                    end
                    if okAll then
                        for _, t in ipairs(tmp) do
                            answers[#answers+1] = t.a
                            breakdown[#breakdown+1] = "  "..t.q:sub(1,26).." = "..t.a
                        end
                    else
                        missing = missing + 1
                        breakdown[#breakdown+1]=part:sub(1,30).." = ?"
                    end
                else
                    missing = missing + 1
                    breakdown[#breakdown+1]=part:sub(1,30).." = ?"
                end
            end
        end
    end
    if #answers == 0 then
        local fz, matched, total
        pcall(function() fz, matched, total = fuzzyResolvePhrase(text) end)
        if fz and #fz > 0 then
            for _, a in ipairs(fz) do answers[#answers+1] = a end
            breakdown[#breakdown+1] = "fuzzy(whole "
                ..tostring(matched or "?").."/"..tostring(total or "?").."): "
                ..table.concat(fz,"+")
            missing = 0
        end
    end
    if #answers==0 then return nil, breakdown, "not in database" end
    local deduped = {}
    for _, a in ipairs(answers) do
        if #deduped == 0 or deduped[#deduped] ~= a then
            deduped[#deduped+1] = a
        end
    end
    return table.concat(deduped,""), breakdown,
        (missing>0) and (missing.." part(s) missing") or nil
end

local ORD_GROUPS = {
    {"1st","first","1"},{"2nd","second","2"},{"3rd","third","3"},
    {"4th","fourth","4"},{"5th","fifth","5"},{"6th","sixth","6"},
    {"7th","seventh","7"},{"8th","eighth","8"},{"9th","ninth","9"},
    {"10th","tenth","10"},{"11th","eleventh","11"},{"12th","twelfth","12"},
    {"13th","thirteenth","13"},{"14th","fourteenth","14"},
}
local function wordBound(t,w) return t:find("%f[%a]"..w.."%f[%A]") ~= nil end
local function replaceOnce(t,f,to) return (t:gsub("%f[%a]"..f.."%f[%A]", to, 1)) end
local function generateAliases(q)
    q = q:lower():gsub("^%s+",""):gsub("%s+$","")
    if q=="" then return {} end
    local out, seen = {}, {[q]=true}
    local function add(a)
        if not a then return end
        a = a:gsub("^%s+",""):gsub("%s+$",""):gsub("%s+"," ")
        if a=="" or #a>60 or seen[a] then return end
        seen[a]=true; out[#out+1]=a
    end
    local ordHit
    for _, grp in ipairs(ORD_GROUPS) do
        for _, o in ipairs(grp) do
            if wordBound(q,o) then ordHit={grp=grp,word=o}; break end
        end
        if ordHit then break end
    end
    if ordHit then
        for _, alt in ipairs(ordHit.grp) do
            if alt ~= ordHit.word then
                add(replaceOnce(q, ordHit.word, alt))
                add(replaceOnce(q, ordHit.word, "the "..alt))
            end
        end
    end
    local intent, subj = q:match("^what%s+(%a+)%s+is%s+(.+)$")
    if intent then
        add(intent.." of "..subj)
        add(subj.." "..intent)
    end
    if q:find("whats",1,true) then add((q:gsub("whats","what is",1))) end
    if q:find("what is",1,true) then add((q:gsub("what is","whats",1))) end
    while #out>20 do table.remove(out) end
    return out
end

local function splitTeachQuestion(q)
    local tmp = q:lower()
    tmp = tmp:gsub("%s*,%s*", " and ")
    tmp = tmp:gsub("%s+then%s+", " and ")
    tmp = tmp:gsub("%s+and%s+", "\1")
    local parts = {}
    for p in tmp:gmatch("[^\1]+") do
        p = trim(p)
        if p ~= "" then parts[#parts+1] = p end
    end
    return parts
end

local function teachRiddle(question, answers)
    if not question or question=="" or not answers or #answers==0 then return false end
    local q = trim(question):lower()
    local cleaned = {}
    for _, a in ipairs(answers) do
        local up = upperClean(a)
        if up ~= "" then cleaned[#cleaned+1]=up end
    end
    if #cleaned==0 then return false end
    local qparts = splitTeachQuestion(q)
    if #qparts >= 2 and #cleaned == #qparts then
        for i = 1, #qparts do
            local subQ = trim(qparts[i]):lower()
            local subA = cleaned[i]
            if subQ ~= "" and not SAB_DB[subQ] then
                SAB_DB[subQ] = subA
                learnedRiddles[subQ] = subA
                for _, al in ipairs(generateAliases(subQ)) do
                    if not SAB_DB[al] then
                        SAB_DB[al] = subA
                        learnedRiddles[al] = subA
                        aliasMeta[al] = subQ
                    end
                end
            end
        end
        local composite = table.concat(cleaned)
        learnedRiddles[q] = composite
        SAB_DB[q] = composite
        for _, al in ipairs(generateAliases(q)) do
            if not SAB_DB[al] then
                SAB_DB[al] = composite
                learnedRiddles[al] = composite
                aliasMeta[al] = q
            end
        end
        saveLearned()
        return true
    end
    learnedRiddles[q]=cleaned[1]; SAB_DB[q]=cleaned[1]
    if #cleaned>1 then
        learnedAlternatives[q]={}
        for i=2,#cleaned do learnedAlternatives[q][#learnedAlternatives[q]+1]=cleaned[i] end
    end
    for _, al in ipairs(generateAliases(q)) do
        if not SAB_DB[al] then
            SAB_DB[al]=cleaned[1]
            learnedRiddles[al]=cleaned[1]
            aliasMeta[al]=q
        end
    end
    saveLearned()
    return true
end

local function forgetRiddle(q)
    q = q:lower()
    if not learnedRiddles[q] then return false end
    local toKill = {q}
    for a,s in pairs(aliasMeta) do if s==q then toKill[#toKill+1]=a end end
    for _, k in ipairs(toKill) do
        SAB_DB[k]=nil; learnedRiddles[k]=nil; aliasMeta[k]=nil; learnedAlternatives[k]=nil
    end
    saveLearned()
    return true
end
_G.addRiddle = function(q,a) return teachRiddle(q,{a}) end
_G.removeRiddle = forgetRiddle

local function installRedeemHook()
    if capturedRedeemRemote then return end

    if not mt or not mt.__namecall then return end
    local old = mt.__namecall
    if setreadonly then pcall(setreadonly, mt, false) end
    mt.__namecall = newcclosure(function(self, ...)
        local m = getnamecallmethod and getnamecallmethod()
        
        if m == "FireServer" and redeemInProgress and not capturedRedeemRemote then
            if typeof(self) == "Instance" and self:IsA("RemoteEvent") then
                local isString = false
                for _, a in ipairs({...}) do
                    if type(a) == "string" and #a >= 1 then isString = true; break end
                end
                
                if isString then
                    capturedRedeemRemote = self
                end
            end
        end
        
        return old(self, ...)
    end)
    if setreadonly then pcall(setreadonly, mt, true) end
end

local function isOurGui(inst)
    local p = inst
    for _=1,10 do
        if not p then break end

local function isVisibleChain(inst)
    local c = inst
    while c do
        local success, isGui = pcall(function() return c:IsA("GuiObject") end)
        if success and isGui then
            local success2, vis = pcall(function() return c.Visible end)
            if success2 and not vis then return false end
        end
        
        local success3, isScreen = pcall(function() return c:IsA("ScreenGui") end)
        if success3 and isScreen then
            local success4, en = pcall(function() return c.Enabled end)
            return success4 and en
        end
        c = c.Parent
    end
    return true
end

local function clearBooleans(fn, depth)
    if not (fn and setupvalue and getupvalues) then return end
    depth = depth or 1
    local ok, ups = pcall(getupvalues, fn)
    if ok and type(ups) == "table" then
        for k, v in pairs(ups) do
            if type(v) == "boolean" then
                pcall(setupvalue, fn, k, false)
            elseif type(v) == "function" and depth > 0 then
                clearBooleans(v, depth - 1)
            end
        end
    end
end

local function fireConns(sig, ...)
    if typeof(getconns) ~= "function" then return false end
    local ok, cs = pcall(getconns, sig)
    if not ok or type(cs) ~= "table" then return false end
    local args = table.pack(...)
    local fired = false
    for _, c in ipairs(cs) do
        local ok2 = pcall(function()
            if c.Enabled ~= false then
                local cb
                pcall(function() cb = c.Function end)
                clearBooleans(cb)
                
                if cb then
                    pcall(task.spawn, cb, table.unpack(args,1,args.n))
                else
                    pcall(function() c:Fire(table.unpack(args,1,args.n)) end)
                end
            end
        end)
        fired = fired or ok2
    end
    return fired
end

local function clickButton(btn)
    if not btn then return false end
    pcall(function() btn.MouseButton1Click:Fire() end)
    pcall(function() btn.Activated:Fire() end)
    if typeof(firesignal) == "function" then
        pcall(firesignal, btn.MouseButton1Click)
        pcall(firesignal, btn.Activated)
    end
    fireConns(btn.MouseButton1Click)
    fireConns(btn.Activated)
    if typeof(fireclick) == "function" then
        pcall(fireclick, btn)
    end
    return true
end
local function fireBoxFocusLost(box)
    if not box then return false end

    end
    if typeof(getconns) == "function" then

        if ok and type(cs) == "table" then
            for _, c in ipairs(cs) do
                pcall(function() if c.Enabled ~= false then c:Fire(true) end end)
            end
        end
    end
    return true
end

local function findAndCacheUI()
    if cacheValid() then return true end
    invalidateCache()
    forceOpenCodesGui()
    local pg = PlayerGui or LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return false end
    local codes = pg:FindFirstChild("Codes")
    local roots = {}
    if codes then roots[#roots+1] = codes end
    roots[#roots+1] = pg
    for _, root in ipairs(roots) do
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("TextBox") and not isOurGui(d) then
                local n = d.Name:lower()
                local pn = (d.Parent and d.Parent.Name or ""):lower()
                if n:find("code") or pn:find("code") or n:find("input")
                    or n:find("redeem") or pn:find("redeem") or n:find("textbox") then
                    if isVisibleChain(d) or (codes and d:IsDescendantOf(codes)) then
                        cachedBox = d; break
                    end
                end
            end
        end
        if cachedBox then break end
    end
    if not cachedBox then
        for _, d in ipairs(pg:GetDescendants()) do
            if d:IsA("TextBox") and not isOurGui(d) and isVisibleChain(d) then
                cachedBox = d; break
            end
        end
    end
    if not cachedBox then return false end
    local node = cachedBox.Parent
    for _ = 1, 8 do
        if not node then break end
        for _, d in ipairs(node:GetDescendants()) do
            if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) and d ~= cachedBox then
                local n = d.Name:lower()
                local t = ""; pcall(function() t = d.Text:lower() end)
                if n:find("submit") or t:find("submit") or n:find("redeem") or t:find("redeem")
                    or n:find("claim") or t:find("claim") or n:find("confirm") or t:find("confirm")
                    or n:find("enter") or t:find("enter") then
                    if isVisibleChain(d) or (codes and d:IsDescendantOf(codes)) then
                        cachedBtn = d; break
                    end
                end
            end
        end
        if cachedBtn then break end
        node = node.Parent
    end
    return true
end

local function fastRedeem(code)
    code = tostring(code or "")
    if code == "" then return end

    redeemInProgress = true

    if capturedRedeemRemote then
        pcall(function() capturedRedeemRemote:FireServer(code) end)
    end

    if not cachedBox then
        findAndCacheUI()
        task.wait(0.15)
        findAndCacheUI()
    end

    task.spawn(function()
        for i = 1, 5 do
            if cachedBox then
                pcall(function() cachedBox.Text = "" end)
                pcall(function() cachedBox.Text = code end)
            end
            if cachedBtn then
                clickButton(cachedBtn)
                fireBoxFocusLost(cachedBox)
            elseif cachedBox then
                fireBoxFocusLost(cachedBox)
            end
            task.wait(0.05)
        end
        task.delay(0.5, function() redeemInProgress = false end)
    end)
end

local function findSimilarRiddles(query, limit)
    if not query or query == "" then return {} end
    local q = basicClean(query)
    if q == "" then return {} end
    local qWords = {}
    for w in q:gmatch("%S+") do
        if #w >= 2 then qWords[#qWords+1] = w end
    end
    local results = {}
    for key, val in pairs(SAB_DB) do
        if #key >= 3 and type(val) == "string" then
            local kl = key:lower()
            local score = 0
            if kl:find(q, 1, true) then score = score + 12 end
            if q:find(kl, 1, true) then score = score + 6 end
            for _, w in ipairs(qWords) do
                if kl:find(w, 1, true) then score = score + 2 end
            end
            if score > 0 then
                results[#results+1] = {key=key, val=val, score=score}
            end
        end
    end
    table.sort(results, function(a,b) return a.score > b.score end)
    while #results > limit do table.remove(results) end
    return results
end

local seenWords = {}

local function currentBox()
    local pg = PlayerGui
    if not pg then return nil end
    local codes = pg:FindFirstChild("Codes")
    if codes then
        local root = codes:FindFirstChild("Codes") or codes
        local redeem = root:FindFirstChild("CodeRedeem")
        local box = redeem and redeem:FindFirstChild("TextBox")
        if box and box:IsA("TextBox") then return box end
        for _, d in ipairs(codes:GetDescendants()) do
            if d:IsA("TextBox") and not isOurGui(d) then return d end
        end
    end
    return nil
end

local NOISE_TOKENS = {"sfx", "itop"}
local function isNoise(text)
    if not text or text == "" then return true end
    local l = text:lower()
    for _, tok in ipairs(NOISE_TOKENS) do
        if l:find(tok, 1, true) then return true end
    end
    return false
end

local function looksLikeCode(text)
    if type(text) ~= "string" then return false end
    if #text < 1 or #text > 30 then return false end
    if not text:match("^[%w_]+$") then return false end
    return true
end

local function pushWord(word)
    if not word or word == "" then return end
    if seenWords[word] then return end
    seenWords[word] = true
    task.delay(1.25, function() seenWords[word] = nil end)

    capturedParts[#capturedParts+1] = word
    local combined = table.concat(capturedParts)
    
    if cfg.spamMode then
        addLog("Captured: "..word.." | Trying: "..combined, C.Green)
        local box = currentBox()
        if box then pcall(function() box.Text = combined end) end
        fastRedeem(combined)
    else
        addLog("Captured "..#capturedParts.."/"..baseThreshold..": "..word, C.Green)
        updateStatusBar()

        local box = currentBox()
        if box then pcall(function() box.Text = combined end) end

        if #capturedParts >= baseThreshold then
            fastRedeem(combined)
            capturedParts = {}
            updateStatusBar()
        end
    end
end

local function hasCodeTrigger(text)
    local l = text:lower()
    return l:find("use code", 1, true) ~= nil
        or l:find("code is", 1, true) ~= nil
        or l:find("new code", 1, true) ~= nil
        or l:find("code:", 1, true) ~= nil
        or l:find("code =", 1, true) ~= nil
end

local function onAnnouncement(raw)
    if type(raw) ~= "string" then return end
    local text = raw:gsub("<[^>]->","")
    text = trim(text)
    if text == "" then return end
    if isNoise(text) then
        addLog("(filtered: "..text:sub(1,30)..")", C.Muted)
        return
    end
    
    addLog("IN: "..text, C.White)

    if hasCodeTrigger(text) then
        codeTriggerArmed = true
        capturedParts = {}
        updateStatusBar()
        if cfg.autoSnipeOnCodeIs and not cfg.codeSniper then
            cfg.codeSniper = true; saveConfig()
            toggleVisual(true, sniperBg, sniperKnob)
            addLog("Auto-enabled Code Sniper (buffer cleared)", C.Green)
        end
        if cfg.onlyAfterUseCode then
            addLog("Code trigger armed - now capturing", C.Green)
        end
    end

    if cfg.onlyAfterUseCode and not codeTriggerArmed then
        addLog("  (ignored - awaiting 'use code'/'code is' trigger)", C.Muted)
        return
    end

    if text:find("%s") then
        if cfg.riddleSolver then
            askedCount = askedCount + 1; bumpSolved()
            local ans, breakdown, note = solveRiddle(text)
            if breakdown then
                for _, line in ipairs(breakdown) do
                    addLog("  "..line, line:sub(-1)=="?" and C.Red or C.Muted)
                end
            end
            if ans then
                solvedCount = solvedCount + 1; bumpSolved()
                addLog("  Answer: "..ans, C.Green)
                if cfg.autoSubmit then
                    task.spawn(function()
                        local box = currentBox()
                        if box then pcall(function() box.Text = ans end) end
                        fastRedeem(ans)
                    end)
                else
                    addLog("  (Auto Submit OFF - answer held)", C.Muted)
                end
            else
                addLog("  "..tostring(note or "unknown"), C.Red)
            end
        else
            addLog("  (Riddle Solver is OFF)", C.Muted)
        end
        return
    end

    if not cfg.codeSniper then
        addLog("  (Code Sniper is OFF)", C.Muted)
        return
    end
    if not looksLikeCode(text) then
        addLog("  (skipped: '"..text.."')", C.Muted)
        return
    end
    
    pushWord(text)
end

local notificationRemote = nil
local listenerHooked = false

local function registerRemote(r)
    if listenerHooked then return end
    if not r or not r.Parent then return end
    if not r:IsA("RemoteEvent") then return end
    listenerHooked = true
    notificationRemote = r
    r.OnClientEvent:Connect(function(...)
        local args = {...}
        if #args == 0 then return end
        local first = args[1]
        local text = nil
        if type(first) == "string" then text = first
        elseif type(first) == "table" then
            text = first.Text or first.Message or first.Content or first.Title
        end
        if type(text) == "string" then task.spawn(onAnnouncement, text) end
    end)
end

local function pathController()
    local ok, controller = pcall(function()
        local c = ReplicatedStorage:FindFirstChild("Controllers")
        local n = c and c:FindFirstChild("NotificationController", true)
        if n then return require(n) end
    end)
    if ok and type(controller)=="table" and type(controller.Start)=="function"
        and typeof(getupvalues)=="function" then
        local ok2, values = pcall(getupvalues, controller.Start)
        if ok2 and type(values)=="table" then
            for _, v in pairs(values) do
                if typeof(v)=="Instance" and v:IsA("RemoteEvent") then
                    registerRemote(v); if listenerHooked then return end
                end
            end
        end
    end
end

local function pathNetPackage()
    if listenerHooked then return end
    if not (getinfo and getconns) then return end
    local packages = ReplicatedStorage:FindFirstChild("Packages")
    local net = packages and packages:FindFirstChild("Net")
    if not net then return end
    for _, d in ipairs(net:GetDescendants()) do
        if listenerHooked then return end
        if d:IsA("RemoteEvent") then
            local ok, cs = pcall(getconns, d.OnClientEvent)
            if ok and type(cs)=="table" then
                for _, c in ipairs(cs) do
                    local f, fn = pcall(function() return c.Function end)
                    if f and type(fn)=="function" then
                        local i, info = pcall(getinfo, fn)
                        if i and info and tostring(info.short_src or info.source or "")
                            :find("NotificationController",1,true) then
                            registerRemote(d); break
                        end
                    end
                end
            end
        end
    end
end
